cmake_minimum_required(VERSION 3.25)
project(EndfieldMotorcycle VERSION 0.5.5 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /permissive-)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(Motorcycle SHARED src/plugin.cpp)
target_include_directories(Motorcycle PRIVATE src sdk/include)
target_link_libraries(Motorcycle PRIVATE crypt32 advapi32)
set_target_properties(Motorcycle PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/$<CONFIG>/motorcycle")
add_custom_target(package ALL DEPENDS Motorcycle)
foreach(f mod.ini config.ini motorcycle.lua native-actions.lua wheel.lua icon.png wheel-icon.png zml-package.json THIRD_PARTY_NOTICES.md assets/sidra-bike.zmlmesh assets/Textures.png licenses/CC-BY-4.0.txt)
  get_filename_component(dir "${f}" DIRECTORY)
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/package/$<CONFIG>/motorcycle/${dir}")
  add_custom_command(TARGET package POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${CMAKE_CURRENT_SOURCE_DIR}/mod/${f}" "${CMAKE_BINARY_DIR}/package/$<CONFIG>/motorcycle/${f}")
endforeach()
enable_testing()
add_executable(ContractTests tests/contract_tests.cpp)
target_include_directories(ContractTests PRIVATE src sdk/include)
target_link_libraries(ContractTests PRIVATE crypt32 advapi32)
add_test(NAME contract COMMAND ContractTests "$<TARGET_FILE:Motorcycle>")
set(ZML_LUPA_PATH "" CACHE PATH "Existing optional Lupa installation")
set(ZML_SERVICES_TEST_EXE "" CACHE FILEPATH "Existing loader public black-box services fixture")
if(ZML_LUPA_PATH AND ZML_SERVICES_TEST_EXE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME bundled_asset COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/model_assets.py")
  add_test(NAME behavior COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/behavior_tests.py"
    --lupa-dir "${ZML_LUPA_PATH}" --services "${ZML_SERVICES_TEST_EXE}")
  set_tests_properties(behavior PROPERTIES TIMEOUT 90)
endif()
