cmake_minimum_required(VERSION 3.25)
project(MotorcycleModelConverter LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(UFBX_DIR "" CACHE PATH "Pinned upstream ufbx source directory")
add_library(ufbx STATIC "${UFBX_DIR}/ufbx.c")
target_include_directories(ufbx PUBLIC "${UFBX_DIR}")
add_executable(BikeConverter main.cpp)
target_link_libraries(BikeConverter PRIVATE ufbx)
target_compile_options(BikeConverter PRIVATE /utf-8)
